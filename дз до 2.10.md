import json
import os
import platform
import psutil

def search_data():
    # 1 Имя компьютера (+ символ в конце)
    comp_name = platform.node() + "_"

    # 2 Общая память (в Gb)
    total_memory = round(psutil.virtual_memory().total / (1024 ** 3), 2)

    # 3 Свободная память (в Gb)
    free_memory = round(psutil.virtual_memory().available / (1024 ** 3), 2)

    # 4 Количество процессов
    process_count = len(psutil.pids())

    # 5 Количество потоков
    thread_count = sum(p.num_threads() for p in psutil.process_iter(['num_threads']) if p.info['num_threads'])

    # 6 Занятая память жесткого диска (HDD/HDR) (в Gb)
    disk_used = round(psutil.disk_usage('/').used / (1024 ** 3), 2)

    # Заполнение словаря именно в том порядке, как указано в ТЗ
    data = {
        "Имя компьютера": comp_name,
        "Общая память (Gb)": total_memory,
        "Свободная память (Gb)": free_memory,
        "Количество процессов": process_count,
        "Количество потоков": thread_count,
        "Занятая память жесткого диска": disk_used
    }
    return data

def save_data(data: dict):
    name = "data.json"
    with open(name, "w", encoding="utf-8") as file:
        json.dump(data, file, ensure_ascii=False, indent=4)

def main():
    data = search_data()
    save_data(data)

if __name__ == "__main__":
    main()
